const express = require('express'); const { createProxyMiddleware } = require('http-proxy-middleware'); const path = require('path'); const app = express(); // Serve the static frontend assets from the public folder app.use(express.static(path.join(__dirname, 'public'))); // Securely proxy search engine traffic with explicit path cleanup app.use('/search', createProxyMiddleware({ target: 'https://html.duckduckgo.com', changeOrigin: true, secure: true, pathRewrite: (pathStr, req) => { // This strips away '/search' completely so DuckDuckGo reads it cleanly return pathStr.replace(/^\/search/, ''); }, on: { proxyRes: (proxyRes, req, res) => { res.setHeader('Access-Control-Allow-Origin', '*'); } } })); // Default port layout for Render environment const PORT = process.env.PORT || 10000; app.listen(PORT, '0.0.0.0', () => { console.log(`Ultablue running smoothly on port ${PORT}`); });
